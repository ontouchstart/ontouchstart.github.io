This should work in the playground.

let mut sieve = [true; 4_000_000];
for i in 2..2_000 {
    if sieve[i] {
        let mut j = i * i;
        while j < 4_000_000 {
            sieve[j] = false;
            j += i;
        }
    }
}
for i in 2..1_000 {
    if sieve[i] {
        println!("prime: {}", i);
    }
}
for i in 1..1_000 {
    if sieve[4_000_000 - i] {
        println!("prime: {}", 4_000_000 - i);
    }
}
